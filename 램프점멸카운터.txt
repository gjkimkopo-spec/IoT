        setInterval(timer1, 50);

        let step = 0;
        let counter = 0;

        function timer1() {
            switch (step) {

                case 100:
                    step = 1;
                    break;

                case 1:
                    document.getElementById('lamp1').checked = true;
                    counter++;
                    document.getElementById('txt1').value = counter;
                    step = 2;
                    break;

                case 2:
                    document.getElementById('lamp1').checked = false;
                    step = 1;
                    break;

                case 200:
                    break;

                case 300:
                    counter = 0;
                    document.getElementById('txt1').value = counter;
                    document.getElementById('lamp1').checked = false;
	            counter = 0;
                    break;
            }
        }

        function start() {
            step = 100;
        }

        function stop() {
            step = 200;
        }

        function reset() {
            step = 300;
        }
